CREATE SEQUENCE client_seq
    START WITH 1
    INCREMENT BY 1;

CREATE TABLE client_details (
    id BIGINT PRIMARY KEY,
    client_identifier VARCHAR(20) NOT NULL UNIQUE,
    name VARCHAR(255),
    address VARCHAR(255),
    phone_number VARCHAR(255),
    create_date TIMESTAMP NOT NULL,
    last_update_date TIMESTAMP,
    license_start_datetime TIMESTAMP,
    license_end_datetime TIMESTAMP,
    churn_rate_2_weeks NUMERIC(5, 2),
    churn_rate_4_weeks NUMERIC(5, 2),
    churn_rate_12_weeks NUMERIC(5, 2),
    login_weekly_history VARCHAR(255)
);

    login_count_per_day INTEGER,
